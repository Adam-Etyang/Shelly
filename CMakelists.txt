cmake_minimum_required(VERSION 3.20)

project(Shell)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(Shell

    src/shell.cpp
    src/parser.cpp
    src/process.cpp
    src/cd.cpp
    src/echo.cpp
    src/pwd.cpp
    src/type.cpp
)

target_include_directories(Shell PRIVATE include)
